"""Verify prior tuning files and hosted assets after adding evaluation assets."""
from pathlib import Path
import hashlib
import json
import urllib.request

root = Path.cwd()
prior = root / 'evidence/commons-v3-navigation-tuning-v1-publication'
current = root / 'evidence/commons-v3-navigation-evaluation-v1-publication'
sha = lambda data: hashlib.sha256(data).hexdigest()
def public_read(url):
    request = urllib.request.Request(url, headers={'User-Agent': 'swarm-societies-evidence-preservation-verifier-v1'})
    assert not request.has_header('Authorization')
    with urllib.request.urlopen(request, timeout=60) as stream:
        return stream.read()
manifest = json.loads((prior / 'manifest.json').read_text())
for name, expected in manifest['artifacts_sha256'].items():
    assert sha((root / name).read_bytes()) == expected, name
previous_assets = json.loads((prior / 'release-assets-public.json').read_text())
release = json.loads(public_read('https://api.github.com/repos/ReloadLightly/swarm-societies/releases/tags/commons-v3-navigation-2026-10-07'))
assert release['id'] == previous_assets['release_id']
assert release['target_commitish'] == 'cc9c2fe95a397de567d093daf1e47e1e240fc3a2'
assets = {asset['name']: asset for asset in release['assets']}
rows = []
for previous in previous_assets['assets']:
    asset = assets[previous['name']]
    data = public_read(asset['browser_download_url'])
    assert asset['id'] == previous['id']
    assert len(data) == asset['size'] == previous['size']
    assert sha(data) == previous['sha256']
    rows.append({'id': asset['id'], 'name': asset['name'], 'size': len(data), 'sha256': sha(data),
                 'asset_identity_preserved': True, 'original_bytes_preserved': True})
result = {'version': 'commons-v3-navigation-tuning-preservation-v1', 'status': 'ok',
          'release_id': release['id'], 'release_target_unchanged': True,
          'local_frozen_artifact_count': len(manifest['artifacts_sha256']),
          'all_local_frozen_hashes_unchanged': True, 'authenticated_download': False,
          'all_original_hosted_assets_byte_identical': True, 'assets': rows}
with (current / 'tuning-assets-preserved.json').open('x') as stream:
    json.dump(result, stream, indent=2, sort_keys=True)
    stream.write('\n')
print(json.dumps({'status': 'ok', 'preserved_tuning_assets': len(rows),
                  'local_frozen_artifacts': len(manifest['artifacts_sha256'])}, indent=2))
